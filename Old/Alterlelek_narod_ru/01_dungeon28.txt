    function playNote(freq, wave) {
      if (settings.mel === 0 || freq === 0) return;
      try {
        const ac = getAudio();
        const osc = ac.createOscillator(); const gain = ac.createGain();
        osc.type = (wave === 'dice') ? 'triangle' : (wave || 'triangle');
        osc.frequency.value = freq;
        osc.connect(gain); gain.connect(ac.destination);
        const now = ac.currentTime;
        const wScale = (wave === 'square') ? 0.6 : 1;
        gain.gain.setValueAtTime(0.045 * (settings.mel/100) * wScale, now);
        gain.gain.exponentialRampToValueAtTime(0.001, now + 0.5);
        osc.start(now); osc.stop(now + 0.52);
      } catch (e) {}
    }

    // ---------- ФОНОВЫЙ ГУЛ ----------
    let ambNodes = null;
    let heartbeat = 0;
    function stopAmbient() {
      if (ambNodes) { try { ambNodes.src.stop(); } catch (e) {} ambNodes = null; }
    }
    function buildAmbient() {
      const ac = getAudio();
      const bufferSize = ac.sampleRate * 2;
      const buffer = ac.createBuffer(1, bufferSize, ac.sampleRate);
      const data = buffer.getChannelData(0);
      let last = 0;
      for (let i = 0; i < bufferSize; i++) {
        const white = Math.random()*2 - 1;
        last = (last + 0.02*white) / 1.02;
        data[i] = last * 3.5;
      }
      const src = ac.createBufferSource();
      src.buffer = buffer; src.loop = true;
      const filter = ac.createBiquadFilter();
      filter.type = 'lowpass'; filter.frequency.value = 120;
      const master = ac.createGain();
      master.gain.value = 0.35 * (settings.amb/100);
      const lfo = ac.createOscillator();
      lfo.frequency.value = 0.08;
      const lfoDepth = ac.createGain();
      lfoDepth.gain.value = 0.18 * (settings.amb/100);
      lfo.connect(lfoDepth); lfoDepth.connect(master.gain); lfo.start();
      src.connect(filter); filter.connect(master); master.connect(ac.destination);
      src.start();
      ambNodes = { src, master, lfoDepth };
    }
    function startAmbient() {
      try {
        if (settings.amb === 0) return;
        getAudio();
        if (!ambNodes) buildAmbient();
      } catch (e) {}
    }
    function ensureAmbient() {
      try {
        if (settings.amb === 0) { stopAmbient(); return; }
        if (!audioCtx) { if (gameActive) startAmbient(); return; }
        if (audioCtx.state === 'suspended') audioCtx.resume();
        if (!ambNodes) buildAmbient();
      } catch (e) {}
    }
    function applyVolumes() {
      if (ambNodes) {
        const k = settings.amb / 100;
        ambNodes.master.gain.value = 0.35 * k;
        ambNodes.lfoDepth.gain.value = 0.18 * k;
      }
    }
    function drip() {
      const delay = 9000 + Math.random()*21000;
      setTimeout(() => {
        try {
          if (audioCtx && settings.amb > 0) {
            if (audioCtx.state === 'suspended') audioCtx.resume();
            const o = audioCtx.createOscillator(); const g = audioCtx.createGain();
            o.type = 'sine';
            o.frequency.setValueAtTime(800 + Math.random()*300, audioCtx.currentTime);
            o.frequency.exponentialRampToValueAtTime(300, audioCtx.currentTime + 0.12);
            g.gain.setValueAtTime(0.08 * (settings.amb/100) * THIN_SCALE, audioCtx.currentTime);
            g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.15);
            o.connect(g); g.connect(audioCtx.destination);
            o.start(); o.stop(audioCtx.currentTime + 0.16);
          }
        } catch (e) {}
        drip();
      }, delay);
    }
    function melodyLoop() {
      if (settings.melody === 'dice') {
        // Кости: каждая нота — новый бросок, никогда не повторяется
        if (settings.mel > 0 && audioCtx && Math.random() < 0.8) {
          const note = DICE_SCALE[Math.floor(Math.random()*DICE_SCALE.length)];
          playNote(note, Math.random() < 0.5 ? 'sine' : 'triangle');
        }
      } else {
        const mel = MELODIES[settings.melody] || MELODIES.cave;
        if (settings.mel > 0 && audioCtx) playNote(mel.notes[melodyIdx], mel.wave);
        melodyIdx = (melodyIdx + 1) % mel.notes.length;
      }
      heartbeat++;
      if (heartbeat >= Math.max(4, Math.round(10000 / settings.tempo))) { heartbeat = 0; ensureAmbient(); }
      setTimeout(melodyLoop, settings.tempo);
    }

    let bgCanvas = null;
    function buildCaveBackground() {
      bgCanvas = document.createElement('canvas');
      bgCanvas.width = 320; bgCanvas.height = 320;
      const c = bgCanvas.getContext('2d');
      c.fillStyle = '#26211d'; c.fillRect(0, 0, 320, 320);
      for (let i = 0; i < 160; i++) {
        const x = Math.random()*320, y = Math.random()*320;
        const r = 4 + Math.random()*18;
        c.fillStyle = Math.random() < 0.5 ? 'rgba(64,54,48,0.35)' : 'rgba(28,24,22,0.35)';
        c.beginPath(); c.ellipse(x, y, r, r*(0.5+Math.random()*0.7), Math.random()*Math.PI, 0, 2*Math.PI); c.fill();
      }
      for (let i = 0; i < 14; i++) {
        let x = Math.random()*320, y = Math.random()*320;
        c.strokeStyle = 'rgba(12,10,9,0.5)'; c.lineWidth = 1 + Math.random()*1.5;
        c.beginPath(); c.moveTo(x, y);
        for (let k = 0; k < 5; k++) { x += (Math.random()-0.5)*40; y += (Math.random()-0.5)*40; c.lineTo(x, y); }
        c.stroke();
      }
      c.fillStyle = 'rgba(10,8,8,0.85)';
      for (let i = 0; i < 8; i++) {
        const x = i*40 + Math.random()*20, w = 14 + Math.random()*18, h = 20 + Math.random()*36;
        c.beginPath(); c.moveTo(x-w/2, 0); c.lineTo(x+w/2, 0); c.lineTo(x, h); c.closePath(); c.fill();
      }
      for (let i = 0; i < 6; i++) {
        const x = Math.random()*320, w = 16 + Math.random()*20, h = 14 + Math.random()*30;
        c.beginPath(); c.moveTo(x-w/2, 320); c.lineTo(x+w/2, 320); c.lineTo(x, 320-h); c.closePath(); c.fill();
      }
      const g = c.createRadialGradient(160, 160, 80, 160, 160, 230);
      g.addColorStop(0, 'rgba(0,0,0,0)'); g.addColorStop(1, 'rgba(0,0,0,0.55)');
      c.fillStyle = g; c.fillRect(0, 0, 320, 320);
    }

    const ENEMY_TYPES = ["💀","👻","🐭","🕷","👺"];

    function monsterLikes(monType, itemType) {
      if (monType === "💀") return itemType === 'heart' || itemType === 'star' || itemType === 'blackHeart';
      if (monType === "🐭") return itemType === 'coin' || itemType === 'shield';
      if (monType === "👻") return itemType === 'heart' || itemType === 'scroll';
      if (monType === "🕷") return itemType === 'shield';
      if (monType === "👺") return true;
      return false;
    }

    let fadeTimer = null;
    function startFadeOut() {
      if (fadeTimer) return;
      fadeTimer = setInterval(() => {
        if (items.length === 0) {
          clearInterval(fadeTimer); fadeTimer = null;
          draw();
          return;
        }
        const removeCount = Math.max(1, Math.ceil(items.length / 40));
        items.splice(0, removeCount);
        draw();
      }, 100);
    }
    function stopFade() {
      if (fadeTimer) { clearInterval(fadeTimer); fadeTimer = null; }
    }

    function cellFree(mx, my, monstersArr) {
      if (player.x === mx && player.y === my) return false;
      if (stairs.x === mx && stairs.y === my && !stairs.active) return false;
      return !monstersArr.some(o => o.x === mx && o.y === my);
    }

    function placeMonster(type, hp, dmg, behavior) {
      let mx, my, ok = false;
      for (let pass = 0; pass < 2 && !ok; pass++) {
        for (let tries = 0; tries < 300 && !ok; tries++) {
          mx = Math.floor(Math.random()*GRID)*CELL;
          my = Math.floor(Math.random()*GRID)*CELL;
          if (pass === 0) {
            const near = Math.abs(mx - player.x) <= 64 && Math.abs(my - player.y) <= 64;
            if (near) continue;
          }
          if (cellFree(mx, my, monsters)) ok = true;
        }
      }
      if (!ok) return;
      monsters.push({
        x: mx, y: my, hp: hp, maxHp: hp, atk: dmg,
        def: (type === "👺" ? 1 : 0),
        isStunned: false, stunTimer: 0,
        type: type, behavior: behavior
      });
    }

    function initFloor() {
      stopFade();
      monsters = [];
      items = [];
      stairs.active = false;
      player.x = 32;
      player.y = 160;

      const count = Math.min(25, currentFloor + settings.monstAdd);
      for (let i = 0; i < count; i++) {
        const type = ENEMY_TYPES[i % ENEMY_TYPES.length];
        const base = ENEMY_BASE[type];
        let hp = base.hp + 10*(currentFloor-1);
        let dmg = base.atk + (currentFloor-1);
        if (settings.difficulty === 'horror') {
          hp = Math.round(hp * 1.5);
          dmg = dmg + 2;
        }
        let behavior = 'hunt';
        if (gameMode === 'chase') behavior = 'flee';
        if (gameMode === 'mixed') behavior = (i % 2 === 0) ? 'flee' : 'hunt';
        placeMonster(type, hp, dmg, behavior);
      }
      updateHud();
    }

    function findFreeCell() {
      let x, y, tries = 0;
      do {
        x = Math.floor(Math.random()*GRID)*CELL;
        y = Math.floor(Math.random()*GRID)*CELL;
        tries++;
      } while (
        (monsters.some(m => m.x===x && m.y===y) ||
         items.some(i => i.x===x && i.y===y) ||
         (player.x===x && player.y===y) ||
         (stairs.x===x && stairs.y===y)) && tries < 200
      );
      return {x, y};
    }

    const ITEM_TYPES = ['coin', 'star', 'shield', 'heart', 'scroll'];
    function spawnItems() {
      if (stairs.active) return;
      if (totalTurns % settings.itemRate === 0) {
        const t = ITEM_TYPES[Math.floor(Math.random()*ITEM_TYPES.length)];
        const cell = findFreeCell();
        items.push({x:cell.x, y:cell.y, type:t});
      }
      while (items.length > MAX_ITEMS) items.shift();
    }

    function openStairs() {
      if (monsters.length === 0) {
        if (!stairs.active) { stairs.active = true; playSound('stairs'); }
        startFadeOut();
      }
    }

    function teleportMonster(m) {
      const dist = Math.floor(Math.random()*3) + 1;
      const options = [];
      if (m.y - dist*CELL >= 0)    options.push(1);
      if (m.y + dist*CELL < 320)   options.push(2);
      if (m.x - dist*CELL >= 0)    options.push(3);
      if (m.x + dist*CELL < 320)   options.push(4);
      if (options.length === 0) return;
      const side = options[Math.floor(Math.random()*options.length)];
      if (side === 1) m.y -= dist*CELL;
      if (side === 2) m.y += dist*CELL;
      if (side === 3) m.x -= dist*CELL;
      if (side === 4) m.x += dist*CELL;
      playSound('tele');
    }

    function fight(m, index) {
      playSound('hit');
      let pDmg = (weapons[weaponLevel].dmg + starBonus) - m.def;
      if (pDmg < 1) pDmg = 1;
      m.hp -= pDmg;
      if (m.hp <= 0) {
        monsters.splice(index, 1);
        goldCount += 2; energy += 1;
        playSound('kill');
        openStairs();
        return;
      }
      if (!m.isStunned) {
        const mDmg = Math.max(0, m.atk - shieldBonus);
        if (mDmg > 0) {
          playerHp -= mDmg;
          if (m.type === "👺" && goldCount > 0) { goldCount--; playSound('steal'); }
        }
      } else {
        m.stunTimer--;
        if (m.stunTimer <= 0) m.isStunned = false;
      }
      if (playerHp <= 0) endGame(false);
    }

    function endGame(win) {
      gameActive = false;
      stopFade();
      if (win) { playSound('win'); alert("ПОБЕДА, " + heroName + "! Пройдено этажей: " + settings.floors + ", ходов: " + totalTurns); }
      else { playSound('lose'); alert(heroName + " пал в бою! Сделано ходов: " + totalTurns); }
      if (totalTurns > highScore) {
        highScore = totalTurns; highScoreEl.innerText = highScore;
        try { localStorage.setItem('pz_highscore', String(highScore)); } catch (e) {}
      }
      goldCount = 0; playerHp = HEROES[heroIdx].maxHp; clearBonuses();
      energy = 0; weaponLevel = 0; currentFloor = 1; totalTurns = 0;
      showMenu();
    }

    function trySmartStep(m) {
      if (!isSmart()) return false;
      for (const it of items) {
        if (!monsterLikes(m.type, it.type)) continue;
        const dx = it.x - m.x, dy = it.y - m.y;
        if (Math.abs(dx) + Math.abs(dy) !== CELL) continue;
        const free = (x, y) => !monsters.some(o => o !== m && o.x === x && o.y === y);
        if (dx !== 0) {
          const nx = m.x + Math.sign(dx)*CELL;
          if (nx >= 0 && nx < 320 && free(nx, m.y)) { m.x = nx; return true; }
        } else {
          const ny = m.y + Math.sign(dy)*CELL;
          if (ny >= 0 && ny < 320 && free(m.x, ny)) { m.y = ny; return true; }
        }
      }
      return false;
    }

    function moveMonsters() {
      if (!gameActive) return;
      for (let idx = monsters.length - 1; idx >= 0; idx--) {
        const m = monsters[idx];
        if (m.isStunned) {
          m.stunTimer--;
          if (m.stunTimer <= 0) m.isStunned = false;
          continue;
        }
        if (m.x === player.x && m.y === player.y) {
          fight(m, idx);
          if (!gameActive) return;
          if (Math.random() < 0.5) teleportMonster(m);
          continue;
        }
        const steps = (m.type === "🕷") ? 2 : 1;
        for (let s = 0; s < steps; s++) {
          if (trySmartStep(m)) {
            if (m.x === player.x && m.y === player.y) {
              fight(m, idx);
              if (!gameActive) return;
            }
            continue;
          }
          const teleportChance = (m.behavior === 'flee') ? 0.35 : 0.25;
          if (Math.random() < teleportChance) {
            teleportMonster(m);
          } else {
            const dx = player.x - m.x, dy = player.y - m.y;
            const sign = (m.behavior === 'flee') ? -1 : 1;
            const sx = (dx > 0 ? 32 : (dx < 0 ? -32 : 0)) * sign;
            const sy = (dy > 0 ? 32 : (dy < 0 ? -32 : 0)) * sign;
            const nx = m.x + sx, ny = m.y + sy;
            const free = (x, y) => !monsters.some(o => o !== m && o.x === x && o.y === y);

            let moved = false;
            if (Math.abs(dx) >= Math.abs(dy)) {
              if (sx !== 0 && nx >= 0 && nx < 320 && free(nx, m.y)) { m.x = nx; moved = true; }
              else if (sy !== 0 && ny >= 0 && ny < 320 && free(m.x, ny)) { m.y = ny; moved = true; }
            } else {
              if (sy !== 0 && ny >= 0 && ny < 320 && free(m.x, ny)) { m.y = ny; moved = true; }
              else if (sx !== 0 && nx >= 0 && nx < 320 && free(nx, m.y)) { m.x = nx; moved = true; }
            }
            if (!moved) teleportMonster(m);
          }

          if (m.x === player.x && m.y === player.y) {
            fight(m, idx);
            if (!gameActive) return;
            if (Math.random() < 0.5) teleportMonster(m);
            break;
          }
        }
      }
    }

    function enemiesEatItems() {
      for (let i = items.length - 1; i >= 0; i--) {
        const it = items[i];
        const mon = monsters.find(m => m.x === it.x && m.y === it.y);
        if (!mon) continue;
        if (mon.type === "💀" && it.type === 'heart') { it.type = 'blackHeart'; playSound('corrupt'); }
        else if (mon.type === "💀" && (it.type === 'star' || it.type === 'blackHeart')) { mon.atk += 1; items.splice(i, 1); playSound('eat'); }
        else if (mon.type === "🐭" && it.type === 'coin') { mon.def += 1; items.splice(i, 1); playSound('eat'); }
        else if (mon.type === "🐭" && it.type === 'shield') { mon.def += 2; items.splice(i, 1); playSound('eat'); }
        else if (mon.type === "👻" && it.type === 'heart') { mon.hp += 20; items.splice(i, 1); playSound('eat'); }
        else if (mon.type === "👻" && it.type === 'scroll') { items.splice(i, 1); playSound('eat'); }
        else if (mon.type === "🕷" && it.type === 'shield') { mon.def += 1; items.splice(i, 1); playSound('eat'); }
        else if (mon.type === "👺") { items.splice(i, 1); playSound('eat'); }
      }
    }

    function afterTurn() {
      if (!gameActive) return;
      totalTurns++;
      spawnItems();
      let pickupIdx = items.findIndex(i => i.x === player.x && i.y === player.y);
      if (pickupIdx !== -1) {
        const item = items[pickupIdx];
        if (item.type === 'coin') goldCount += 2;
        if (item.type === 'star') starBonus += settings.starVal;
        if (item.type === 'shield') shieldBonus += settings.shieldVal;
        if (item.type === 'heart') playerHp += 20;
        if (item.type === 'scroll') energy += 2;
        if (item.type === 'blackHeart') { playerHp -= 15; playSound('bad'); }
        else playSound('collect');
        items.splice(pickupIdx, 1);
        if (playerHp <= 0) { endGame(false); return; }
      }
      moveMonsters();
      if (!gameActive) return;
      enemiesEatItems();
      updateHud();
      draw();
    }

    function move(dx, dy) {
      if (!gameActive || playerHp <= 0 || shopOpen || settingsOpen) return;
      let nx = player.x + dx, ny = player.y + dy;
      if (nx < 0 || nx >= 320 || ny < 0 || ny >= 320) return;
      player.x = nx; player.y = ny;
      playSound('step');
      if (stairs.active && nx === stairs.x && ny === stairs.y) {
        currentFloor++;
        if (currentFloor > settings.floors) { endGame(true); return; }
        playSound('stairs');
        if (settings.bonusReset) clearBonuses();
        initFloor();
        draw();
        return;
      }
      afterTurn();
    }

    function skipTurn() {
      if (!gameActive || playerHp <= 0 || shopOpen || settingsOpen) return;
      const victimIdx = monsters.findIndex(m => m.x === player.x && m.y === player.y);
      if (victimIdx !== -1) {
        fight(monsters[victimIdx], victimIdx);
        if (!gameActive) return;
      } else playSound('skip');
      afterTurn();
    }

    function castLightning() {
      if (!gameActive || energy < lightCost() || shopOpen || settingsOpen) return;
      energy -= lightCost();
      playSound('magic');
      const range = HEROES[heroIdx].lightRange;
      let killed = 0;
      for (let i = monsters.length - 1; i >= 0; i--) {
        const dx = Math.abs(monsters[i].x - player.x);
        const dy = Math.abs(monsters[i].y - player.y);
        if (dx <= range && dy <= range) {
          monsters[i].hp -= 20;
          if (monsters[i].hp <= 0) { monsters.splice(i, 1); killed++; playSound('kill'); }
        }
      }
      if (killed > 0) { goldCount += killed*2; energy += killed; openStairs(); }
      updateHud(); draw();
    }

    function castStun() {
      if (!gameActive || energy < stunCost() || shopOpen || settingsOpen) return;
      energy -= stunCost();
      playSound('magic');
      monsters.forEach(m => { m.isStunned = true; m.stunTimer = 2; });
      updateHud(); draw();
    }

    function drawCell(x, y, emoji, size) {
      ctx.font = (size || 20) + "px sans-serif";
      ctx.textAlign = "center"; ctx.textBaseline = "middle";
      ctx.fillText(emoji, x + 16, y + 17);
    }
    function drawBars(x, y, hp, maxHp, def, atk) {
      const frac = Math.max(0, Math.min(1, hp / maxHp));
      ctx.fillStyle = "#4b5563"; ctx.fillRect(x+4, y+1, 24, 3);
      ctx.fillStyle = frac > 0.5 ? "#22c55e" : "#ef4444";
      ctx.fillRect(x+4, y+1, 24*frac, 3);
      ctx.fillStyle = def > 0 ? "#3b82f6" : "#4b5563";
      ctx.beginPath(); ctx.arc(x+7, y+25, 6, 0, 2*Math.PI); ctx.fill();
      ctx.strokeStyle = "#1e3a8a"; ctx.lineWidth = 1; ctx.stroke();
      ctx.fillStyle = "#fff"; ctx.font = "8px sans-serif";
      ctx.textAlign = "center"; ctx.textBaseline = "middle";
      ctx.fillText(def, x+7, y+26);
      ctx.fillStyle = "#dc2626";
      ctx.beginPath(); ctx.arc(x+25, y+25, 6, 0, 2*Math.PI); ctx.fill();
      ctx.strokeStyle = "#7f1d1d"; ctx.stroke();
      ctx.fillText(atk, x+25, y+26);
    }
    function drawItem(x, y, type) {
      const cx = x + 16, cy = y + 16;
      if (type === 'coin') {
        ctx.fillStyle = "#fbbf24"; ctx.beginPath(); ctx.arc(cx, cy, 8, 0, 2*Math.PI); ctx.fill();
        ctx.strokeStyle = "#92400e"; ctx.lineWidth = 2; ctx.stroke();
        ctx.fillStyle = "#fde68a"; ctx.beginPath(); ctx.arc(cx-2, cy-2, 3, 0, 2*Math.PI); ctx.fill();
      } else if (type === 'star') {
        ctx.fillStyle = "#fbbf24"; ctx.beginPath();
        for (let k = 0; k < 5; k++) {
          const a1 = -Math.PI/2 + k*2*Math.PI/5, a2 = a1 + Math.PI/5;
          ctx.lineTo(cx + Math.cos(a1)*9, cy + Math.sin(a1)*9);
          ctx.lineTo(cx + Math.cos(a2)*4, cy + Math.sin(a2)*4);
        }
        ctx.closePath(); ctx.fill();
      } else if (type === 'shield') {
        ctx.fillStyle = "#3b82f6"; ctx.beginPath();
        ctx.moveTo(cx-7, cy-8); ctx.lineTo(cx+7, cy-8); ctx.lineTo(cx+7, cy+2);
        ctx.quadraticCurveTo(cx, cy+10, cx, cy+10); ctx.quadraticCurveTo(cx-7, cy+2, cx-7, cy+2);
        ctx.closePath(); ctx.fill();
        ctx.strokeStyle = "#1e3a8a"; ctx.lineWidth = 1.5; ctx.stroke();
      } else if (type === 'heart' || type === 'blackHeart') {
        ctx.fillStyle = (type === 'heart') ? "#ef4444" : "#111111";
        ctx.beginPath();
        ctx.moveTo(cx, cy+7);
        ctx.bezierCurveTo(cx-11, cy-2, cx-7, cy-10, cx, cy-4);
        ctx.bezierCurveTo(cx+7, cy-10, cx+11, cy-2, cx, cy+7);
        ctx.closePath(); ctx.fill();
        if (type === 'blackHeart') { ctx.strokeStyle = "#ef4444"; ctx.lineWidth = 1; ctx.stroke(); }
      } else if (type === 'scroll') {
        ctx.fillStyle = "#e5d3a1"; ctx.fillRect(cx-7, cy-8, 14, 16);
        ctx.fillStyle = "#7c3aed"; ctx.fillRect(cx-7, cy-8, 14, 4); ctx.fillRect(cx-7, cy+4, 14, 4);
        ctx.fillStyle = "#6b7280"; ctx.fillRect(cx-4, cy-2, 8, 1.5); ctx.fillRect(cx-4, cy+1, 8, 1.5);
      }
    }
    function draw() {
      if (bgCanvas) ctx.drawImage(bgCanvas, 0, 0);
      else { ctx.fillStyle = "#241f1d"; ctx.fillRect(0, 0, 320, 320); }
      ctx.strokeStyle = "rgba(90,80,70,0.4)"; ctx.lineWidth = 1;
      for (let i = 0; i <= 320; i += 32) {
        ctx.beginPath(); ctx.moveTo(i, 0); ctx.lineTo(i, 320); ctx.stroke();
        ctx.beginPath(); ctx.moveTo(0, i); ctx.lineTo(320, i); ctx.stroke();
      }
      if (stairs.active) {
        ctx.fillStyle = "#f59e0b"; ctx.fillRect(stairs.x+4, stairs.y+4, 24, 24);
        ctx.fillStyle = "#1e1e24"; drawCell(stairs.x, stairs.y, "▼", 16);
      }
      items.forEach(i => drawItem(i.x, i.y, i.type));
      monsters.forEach(m => {
        if (m.isStunned) { ctx.fillStyle = "rgba(147,51,234,0.5)"; ctx.fillRect(m.x+2, m.y+2, 28, 28); }
        drawCell(m.x, m.y, m.type, 22);
        drawBars(m.x, m.y, m.hp, m.maxHp, m.def, m.atk);
      });
      drawCell(player.x, player.y, HEROES[heroIdx].emoji, 22);
      drawBars(player.x, player.y, playerHp, heroMaxHp(), shieldBonus, weapons[weaponLevel].dmg + starBonus);
      const sharing = monsters.filter(m => m.x === player.x && m.y === player.y);
      if (sharing.length > 0) {
        ctx.strokeStyle = "rgba(239,68,68,0.9)"; ctx.lineWidth = 3;
        ctx.strokeRect(player.x+1, player.y+1, 30, 30);
      }
    }

    // ================= УПРАВЛЕНИЕ =================
    function wasdAllowed() { return settings.keysMode !== 'arrows'; }
    function arrowsAllowed() { return settings.keysMode !== 'wasd'; }
    function letterOk(latKey, cyrKey) {
      if (settings.keysLayout === 'both') return true;
      if (settings.keysLayout === 'lat') return !/^[а-яё]$/i.test(cyrKey || '');
      return !/^[a-z]$/i.test(latKey || '');
    }
    document.addEventListener('keydown', (e) => {
      if (e.key === 'Escape') {
        if (settingsOpen) { toggleSettings(false); return; }
        if (shopOpen) { toggleShop(false); return; }
      }
      if (shopOpen || settingsOpen) return;
      if (menuActive) {
        const up = e.key === 'ArrowUp' || (wasdAllowed() && (e.key.toLowerCase() === 'w' && letterOk('w', 'ц')));
        const down = e.key === 'ArrowDown' || (wasdAllowed() && (e.key.toLowerCase() === 's' && letterOk('s', 'ы')));
        if (up) { menuIdx = (menuIdx + 2) % 3; playSound('menu'); updateMenu(); }
        else if (down) { menuIdx = (menuIdx + 1) % 3; playSound('menu'); updateMenu(); }
        else if (e.key === 'Enter' || e.key === ' ') selectMode(menuIdx);
        else if (e.key === '1') selectMode(0);
        else if (e.key === '2') selectMode(1);
        else if (e.key === '3') selectMode(2);
        return;
      }
      const lat = e.key.toLowerCase();
      if ((arrowsAllowed() && e.key === 'ArrowUp') || (wasdAllowed() && lat === 'w' && letterOk('w', 'ц'))) { move(0, -32); return; }
      if ((arrowsAllowed() && e.key === 'ArrowLeft') || (wasdAllowed() && lat === 'a' && letterOk('a', 'ф'))) { move(-32, 0); return; }
      if ((arrowsAllowed() && e.key === 'ArrowDown') || (wasdAllowed() && lat === 's' && letterOk('s', 'ы'))) { move(0, 32); return; }
      if ((arrowsAllowed() && e.key === 'ArrowRight') || (wasdAllowed() && lat === 'd' && letterOk('d', 'в'))) { move(32, 0); return; }
      if (e.key === ' ') skipTurn();
      else if (e.key === '1') castLightning();
      else if (e.key === '3') castStun();
      else if (lat === 'm' || e.key === 'ь') toggleShop(true);
    });

    try {
      const hs = localStorage.getItem('pz_highscore');
      if (hs) highScore = parseInt(hs, 10) || 0;
    } catch (e) {}
    highScoreEl.innerText = highScore;

    loadSettings();
    melodyLoop();
    drip();
    scheduleAutoMelody();
    updateMenu();
  </script>
