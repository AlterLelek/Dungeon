function gridNew(){ return Array.from({length:ROWS},()=>Array(COLS).fill(1)); }
function inGrid(x,y){ return x>0&&y>0&&x<COLS-1&&y<ROWS-1; }
function genMaze(){
 const g=gridNew();
 const stack=[[1,1]]; g[1][1]=0;
 while(stack.length){
  const [x,y]=stack[stack.length-1];
  const dirs=[[2,0],[-2,0],[0,2],[0,-2]].filter(([dx,dy])=>{
   const nx=x+dx,ny=y+dy;
   return nx>0&&ny>0&&nx<COLS-1&&ny<ROWS-1&&g[ny][nx]===1;
  });
  if(dirs.length){ const [dx,dy]=pick(dirs);
   g[y+dy/2][x+dx/2]=0; g[y+dy][x+dx]=0; stack.push([x+dx,y+dy]);
  } else stack.pop();
 }
 for(let i=0;i<COLS*ROWS/7;i++){ const x=irand(1,COLS-2),y=irand(1,ROWS-2);
  if(g[y][x]===1&&((g[y][x-1]===0&&g[y][x+1]===0)||(g[y-1][x]===0&&g[y+1][x]===0)))g[y][x]=0; }
 return g;
}
function genRooms(){
 const g=gridNew(); const rooms=[];
 for(let i=0;i<irand(6,9);i++){
  const w=irand(3,6),h=irand(2,4),x=irand(1,COLS-w-2),y=irand(1,ROWS-h-2);
  rooms.push([x,y,w,h]);
  for(let yy=y;yy<y+h;yy++)for(let xx=x;xx<x+w;xx++)g[yy][xx]=0;
 }
 for(let i=1;i<rooms.length;i++){
  const [x1,y1]=rooms[i-1],[x2,y2]=rooms[i];
  const cx1=x1+2,cy1=y1+1,cx2=x2+2,cy2=y2+1;
  for(let x=Math.min(cx1,cx2);x<=Math.max(cx1,cx2);x++)g[cy1][x]=0;
  for(let y=Math.min(cy1,cy2);y<=Math.max(cy1,cy2);y++)g[y][cx2]=0;
 }
 return g;
}

}

function bfsStep(sx,sy,tx,ty){
 if(sx===tx&&sy===ty)return null;
 const q=[[sx,sy]],prev={},seen=new Set([sx+','+sy]);
 while(q.length){
  const [x,y]=q.shift();
  for(const [dx,dy] of [[1,0],[-1,0],[0,1],[0,-1]]){
   const nx=x+dx,ny=y+dy,k=nx+','+ny;
   if(nx<0||ny<0||nx>=COLS||ny>=ROWS||G.grid[ny][nx]||seen.has(k))continue;
   seen.add(k);prev[k]=[x,y];
   if(nx===tx&&ny===ty){
    let cx=nx,cy=ny;
    while(true){const pp=prev[cx+','+cy];if(pp[0]===sx&&pp[1]===sy)return [cx,cy];cx=pp[0];cy=pp[1];}
   }
   q.push([nx,ny]);
  }
 }
 return null;
}
function monsterTarget(m){
 const cfg=S.monsters[m.type]||{};
 const taste=cfg.taste||m.taste;
 const R=cfg.radius!=null?cfg.radius:m.radius;
 let tx=null,ty=null;
 if(taste==='hero'){tx=G.hero.x;ty=G.hero.y;}
 else if(taste==='gem'&&G.gemList&&G.gemList.length){let bd=99;for(const g of G.gemList){const d=dist(m.x,m.y,g.x,g.y);if(d<bd){bd=d;tx=g.x;ty=g.y;}}}
 else if(taste==='coin'&&G.items&&G.items.length){let bd=99;for(const it of G.items){if(it.kind!=='coin')continue;const d=dist(m.x,m.y,it.x,it.y);if(d<bd){bd=d;tx=it.x;ty=it.y;}}}
 else if(taste==='corpse'&&G.corpses&&G.corpses.length){let bd=99;for(const c of G.corpses){const d=dist(m.x,m.y,c.x,c.y);if(d<bd){bd=d;tx=c.x;ty=c.y;}}}
 if(tx==null)return null;
 if(dist(m.x,m.y,tx,ty)>R)return null;
 return [tx,ty];
}
function monsterTick(){
 const eff=S.pause.monPause;
 const skipped=G.tick%(eff+1)!==0;
 G.tick++;
 if(skipped)return;
 for(const m of G.monsters){
  if(m.dying)continue;
  m.spawnT=Math.min(1,m.spawnT+0.08);
  m.cnt++;
  const slow=m.slow||1;
  if(slow>1&&m.cnt%slow!==0)continue; // медленные: шаг, потом стоят
  let steps=m.steps;
  if(m.type==='spider')steps*=Math.max(1,S.pause.spiderStep);
  const tgt=monsterTarget(m);
  let nx=m.x,ny=m.y,moved=false;
  for(let s=0;s<steps;s++){
   let step=null;
   if(tgt)step=bfsStep(m.x,m.y,tgt[0],tgt[1],99);
   if(!step){ // блуждание
    const dirs=pick([[1,0],[-1,0],[0,1],[0,-1]]);
    const wx=m.x+dirs[0],wy=m.y+dirs[1];
    if(inGrid(wx,wy)&&!G.grid[wy][wx])step=[wx,wy];
   }
   if(step){
    const blocked=G.monsters.some(o=>o!==m&&!o.dying&&o.x===step[0]&&o.y===step[1]);
    const blockedAlly=G.allies.some(a=>!a.dying&&a.x===step[0]&&a.y===step[1]);
    if(!blocked&&!blockedAlly&&!(G.hero.x===step[0]&&G.hero.y===step[1])){
     m.x=step[0];m.y=step[1];moved=true;
    }
   }
  }
  if(dist(m.x,m.y,G.hero.x,G.hero.y)===1&&m.type!=='ghost'){
   const tk=(S.monsters[m.type]||{}).taste||m.taste;
   if(tk==='hero'||Math.random()<0.5)monsterAttack(m);
  }
  if(moved)m.animOff=Math.random()*6.28;
 }
 // некромант оживляет
 for(const m of G.monsters){
  if(m.dying||m.type!=='necro')continue;
  m.raiseCnt=(m.raiseCnt||0)+1;
  const period=S.monsters.necro.period||6;
  if(m.raiseCnt>=period){
   m.raiseCnt=0;
   let best=null,bd=999;
   for(const c of G.corpses){const d=dist(m.x,m.y,c.x,c.y);if(d<bd&&d<=S.monsters.necro.radius){bd=d;best=c;}}
   if(best){
    G.corpses.splice(G.corpses.indexOf(best),1);
    const z=makeMonster('zombie',best.x,best.y);
    z.spawnT=0;
    G.monsters.push(z);
    addParts((best.x+0.5)*TS,(best.y+0.5)*TS,'#60ff90',20,70);
    addText((best.x+0.5)*TS,(best.y+0.2)*TS,'ЗОМБИ!','#60ff90');
    SFX.raise();doShake(2);
   }
  }
 }
 // союзники
 for(const a of G.allies){
  if(a.dying)continue;
  a.spawnT=Math.min(1,a.spawnT+0.08);
  a.cnt++;
  if(a.slow>1&&a.cnt%a.slow!==0)continue;
  let bm=null,bd=99;
  for(const m of G.monsters){if(m.dying)continue;const d=dist(a.x,a.y,m.x,m.y);if(d<bd){bd=d;bm=m;}}
  if(!bm)continue;
  if(bd===1){
   const dmg=Math.max(1,Math.round(a.atk*rand(0.8,1.2)-bm.def));
   damageMonster(bm,dmg,'#8fd0ff');
  } else if(a.kind==='imp'){
   a.cool=(a.cool||0)-1;
   if(a.cool<=0&&bd<=6){
    a.cool=6;
    const dx=Math.sign(bm.x-a.x),dy=Math.sign(bm.y-a.y);
    G.projs.push({x:a.x,y:a.y,dx:dx||0,dy:dy||(dx?0:1),dmg:a.atk,kind:'imp',owner:'ally',t:0});
   }
  } else {
   const step=bfsStep(a.x,a.y,bm.x,bm.y,99);
   if(step&&!G.monsters.some(o=>!o.dying&&o.x===step[0]&&o.y===step[1])){a.x=step[0];a.y=step[1];}
  }
 }
 G.corpses=G.corpses.filter(c=>(c.life-=0.5)>0);
}
function projTick(dt){
 for(let i=G.projs.length-1;i>=0;i--){
  const p=G.projs[i];p.t+=dt;
  if(p.t<110)continue;
  p.t=0;
  p.x+=p.dx;p.y+=p.dy;
  if(p.x<0||p.y<0||p.x>=COLS||p.y>=ROWS||G.grid[p.y][p.x]){
   addParts((p.x+0.5)*TS,(p.y+0.5)*TS,p.kind==='fire'?'#ff8030':'#a0c0ff',8,50);
   G.projs.splice(i,1);continue;
  }
  addParts((p.x+0.5)*TS,(p.y+0.5)*TS,p.kind==='fire'?'#ff8030':'#a0c0ff',2,20,0.3);
  let hit=null;
  if(p.owner==='hero'){hit=G.monsters.find(m=>!m.dying&&m.x===p.x&&m.y===p.y);}
  else hit=G.monsters.find(m=>!m.dying&&m.x===p.x&&m.y===p.y);
  if(hit){
   const dmg=Math.max(1,Math.round(p.dmg-hit.def*0.5));
   damageMonster(hit,dmg,p.kind==='fire'?'#ff9040':'#a0c0ff');
   addParts((p.x+0.5)*TS,(p.y+0.5)*TS,'#ff8030',16,80);
   addRing((p.x+0.5)*TS,(p.y+0.5)*TS,TS*0.8,p.kind==='fire'?'#ff8030':'#a0c0ff');
   G.projs.splice(i,1);
  }
 }
}

let lastT=0;
function loop(t){
 requestAnimationFrame(loop);
 const dt=Math.min(60,t-lastT);lastT=t;
 if(G&&G.running&&!G.paused){
  update(dt);
 } else if(G&&G.fxOnly){
  const d2=dt/1000;
  for(let i=G.parts.length-1;i>=0;i--){const p=G.parts[i];p.life-=d2;p.x+=p.vx*d2;p.y+=p.vy*d2;if(p.life<=0)G.parts.splice(i,1);}
  for(let i=G.rings.length-1;i>=0;i--){const r=G.rings[i];r.t+=d2*3;r.r=lerp(4,r.maxR,r.t);if(r.t>=1)G.rings.splice(i,1);}
 }
 if(G)render(t);
}

 h.px=h.x*TS;h.py=h.y*TS;
 // ввод

 G.monTimer+=dt;

 for(let i=G.texts.length-1;i>=0;i--){const x=G.texts[i];x.life-=d*1.4;x.y-=d*26;if(x.life<=0)G.texts.splice(i,1);}

  if(G.fade.done&&G.fade.dir>0){G.fade.dir=-1;}
  if(G.fade.a<=0&&G.fade.done)G.fade=null;
 }
 // таймеры режимов

  if(G.waveT>=20){G.waveT=0;toast('🌊 Новая волна монстров!');
   const free=freeCells(G.grid).filter(c=>dist(c[0],c[1],G.hero.x,G.hero.y)>5);

}

function tileRect(x,y){ return [OFFX+x*TS,OFFY+y*TS,TS,TS]; }
function drawGrid(t){
 const pal=PAL[S.pal]||PAL[0],br=S.palBright/100;
 for(let y=0;y<ROWS;y++)for(let x=0;x<COLS;x++){
  const [px,py]=tileRect(x,y);
  if(G.grid[y][x]){
   ctx.fillStyle=applyBr(pal.wall,br);
   ctx.fillRect(px,py,TS,TS);
   ctx.fillStyle=applyBr(pal.wallTop,br);
   ctx.fillRect(px,py,TS,Math.max(2,TS*0.16));
  } else {
   let a=1;
   if(S.flicker>0)a=1-(S.flicker/100)*0.12*(0.5+0.5*Math.sin(t/350+x*1.7+y*2.3));
   ctx.globalAlpha=a;
   ctx.fillStyle=applyBr(pal.floor,br);
   ctx.fillRect(px,py,TS,TS);
   ctx.globalAlpha=1;
  }
 }
}
function applyBr(hex,br){
 if(br>=0.99)return hex;
 const n=parseInt(hex.slice(1),16),r=(n>>16)&255,g=(n>>8)&255,b=n&255;
 const f=(c)=>clamp(Math.round(c*br),0,255);
 return 'rgb('+f(r)+','+f(g)+','+f(b)+')';
}

 ctx.save();
 ctx.globalAlpha=clamp(alpha,0,1);
 const px=OFFX+x*TS+TS/2, py=OFFY+y*TS+TS/2;
 const s=scale==null?1:scale;
 if(flash>0){
  ctx.globalAlpha=alpha*flash*0.7;
  ctx.fillStyle='#ffffff';
  ctx.beginPath();ctx.arc(px,py,size*0.46*s,0,6.29);ctx.fill();
  ctx.globalAlpha=clamp(alpha,0,1);
 }
 ctx.shadowColor=color;ctx.shadowBlur=size*0.45;
 ctx.font=Math.round(size*s)+'px serif';
 ctx.textAlign='center';ctx.textBaseline='middle';
 ctx.fillText(emoji,px,py+size*0.04);
 ctx.restore();
}

  ctx.globalAlpha=clamp(1-r.t,0,1);
  ctx.strokeStyle=r.color;ctx.lineWidth=3;
  ctx.beginPath();ctx.arc(r.x,r.y,r.r,0,6.29);ctx.stroke();
 }
 ctx.globalAlpha=1;
 ctx.font='bold '+Math.round(TS*0.34)+'px sans-serif';
 ctx.textAlign='center';
 for(const x of G.texts){
  ctx.globalAlpha=clamp(x.life,0,1);
  ctx.strokeStyle='#000';ctx.lineWidth=3;
  ctx.strokeText(x.text,x.x,x.y);
  ctx.fillStyle=x.color;ctx.fillText(x.text,x.x,x.y);
 }
 ctx.globalAlpha=1;
}
function drawFog(){
 if(curMode!=='fog')return;
 for(let y=0;y<ROWS;y++)for(let x=0;x<COLS;x++){
  const [px,py]=tileRect(x,y);
  if(!G.explored.has(x+','+y)){
   ctx.fillStyle='#05050a';ctx.fillRect(px,py,TS,TS);
  } else if(!G.vis.has(x+','+y)){
   ctx.fillStyle='rgba(5,5,12,.55)';ctx.fillRect(px,py,TS,TS);
  }
 }
}
function render(t){
 ctx.setTransform(1,0,0,1,0,0);
 ctx.fillStyle='#0c0c14';ctx.fillRect(0,0,cv.width,cv.height);
 if(!G)return;
 ctx.save();
 if(G.shake>0)ctx.translate(rand(-G.shake,G.shake),rand(-G.shake,G.shake));
 drawGrid(t);
 drawStuff(t);
 drawAllies(t);
 drawMonsters(t);
 drawHero(t);
 drawProjs();
 drawFx();
 drawFog();
 ctx.restore();
 if(G.fade){ctx.fillStyle='rgba(0,0,0,'+clamp(G.fade.a,0,1)+')';ctx.fillRect(0,0,cv.width,cv.height);}
}
